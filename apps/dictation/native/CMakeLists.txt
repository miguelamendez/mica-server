cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(mica_dictation LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  include(FetchContent)
  set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ftxui GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git GIT_TAG v7.0.3 GIT_SHALLOW TRUE)
  FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
  FetchContent_MakeAvailable(ftxui nlohmann_json)
endif()
add_executable(mica-dictation main.cpp)
target_link_libraries(mica-dictation PRIVATE ftxui::component ftxui::dom ftxui::screen)
if(MICA_JSON_INCLUDE_DIR)
  target_include_directories(mica-dictation PRIVATE ${MICA_JSON_INCLUDE_DIR})
else()
  target_link_libraries(mica-dictation PRIVATE nlohmann_json::nlohmann_json)
endif()
find_package(Threads REQUIRED)
target_link_libraries(mica-dictation PRIVATE Threads::Threads)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(mica-dictation PRIVATE platform_macos.mm)
  target_compile_options(mica-dictation PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  target_link_libraries(mica-dictation PRIVATE "-framework AppKit" "-framework AVFoundation" "-framework ApplicationServices")
else()
  target_sources(mica-dictation PRIVATE platform_linux.cpp)
  find_package(X11 QUIET)
  if(X11_FOUND)
    target_compile_definitions(mica-dictation PRIVATE MICA_DICTATION_X11)
    target_link_libraries(mica-dictation PRIVATE X11::X11)
  endif()
endif()
install(TARGETS mica-dictation RUNTIME DESTINATION bin)
